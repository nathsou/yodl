module CharAt<Length: uint, IndexWidth: uint>(
    str: Char[Length],
    index: uint<IndexWidth>,
) -> (
    char: Char,
    is_out_of_bounds: bool,
) {
    is_out_of_bounds = index >= Length;
    char = is_out_of_bounds ? ' ' : str[index];
}

const Rows = VGA::V_ACTIVE / 8;
const Cols = VGA::H_ACTIVE / 8;

module Text<Length: uint, X: uint, Y: uint>(
    text: Char[Length],
    screen_x: uint<10>,
    screen_y: uint<10>,
) -> (
    is_active: bool,
) {
    const StartIdx = Y * Cols + X;
    let x_div = screen_x[9:3];
    let y_div = screen_y[9:3];
    let x = screen_x[2:0];
    let y = screen_y[2:0];
    let cell_idx: uint<13> = y_div * Cols + x_div;
    let diff: uint<13> = cell_idx - StartIdx;
    let char = cell_idx >= StartIdx ? CharAt<Length: Length, IndexWidth: 13>(str: text, index: diff).char : ' ';
    let bitmap = AsciiTable(char).q;
    let pixel_idx: uint<6> = y * 8 + x;
    is_active = (bitmap shr pixel_idx)[0];
}

module TextDynamicPosition<Length: uint>(
    text: Char[Length],
    x_offset: uint<$clog2(Cols)>,
    y_offset: uint<$clog2(Rows)>,
    screen_x: uint<10>,
    screen_y: uint<10>,
) -> (
    is_active: bool,
) {
    let x_div = screen_x[9:3];
    let y_div = screen_y[9:3];
    let x = screen_x[2:0];
    let y = screen_y[2:0];
    let start_idx: uint<13> = y_offset * Cols + x_offset;
    let cell_idx: uint<13> = y_div * Cols + x_div;
    let diff: uint<13> = cell_idx - start_idx;
    let char = cell_idx >= start_idx ? CharAt<Length: Length, IndexWidth: 13>(str: text, index: diff).char : ' ';
    let bitmap = AsciiTable(char).q;
    let pixel_idx: uint<6> = y * 8 + x;
    is_active = (bitmap shr pixel_idx)[0];
}

module Hex<Bits: uint>(n: uint<Bits>) -> (chars: Char[$cdiv(Bits, 4)]) {
    const Len = $cdiv(Bits, 4);
    for i in 0..<Len {
        chars[Len - 1 - i] = (match n[(i + 1) * 4 - 1 -: 4] {
            4'h0 => '0',
            4'h1 => '1',
            4'h2 => '2',
            4'h3 => '3',
            4'h4 => '4',
            4'h5 => '5',
            4'h6 => '6',
            4'h7 => '7',
            4'h8 => '8',
            4'h9 => '9',
            4'hA => 'A',
            4'hB => 'B',
            4'hC => 'C',
            4'hD => 'D',
            4'hE => 'E',
            4'hF => 'F',
        });
    }
}

module Dec<Bits: uint>(n: uint<Bits>) -> (chars: Char[$cdiv(Bits, 4)]) {
    const Len = $cdiv(Bits, 4);
    for i in 0..<Len {
        chars[Len - 1 - i] = (match n[(i + 1) * 4 - 1 -: 4] {
            4'h0 => '0',
            4'h1 => '1',
            4'h2 => '2',
            4'h3 => '3',
            4'h4 => '4',
            4'h5 => '5',
            4'h6 => '6',
            4'h7 => '7',
            4'h8 => '8',
            _ => '9',
        });
    }
}

module Bin<Bits: uint>(n: uint<Bits>) -> (chars: Char[Bits]) {
    for i in 0..<Bits {
        chars[Bits - 1 - i] = n[i] ? '1' : '0';
    }
}

Now we can use modules from the Text package to display a simple some scrolling messages on the screen with a random colour.

import Timing
import LFSR
import Text

module Top(
    clk: clock,
) -> (
    led: uint<4>,
    vga_hsync: bool,
    vga_vsync: bool,
    vga_red: uint<3>,
    vga_green: uint<3>,
    vga_blue: uint<3>,
) {
    let vga = VGA::SyncPulses(
        pixel_clk: clk,
        hsync: vga_hsync,
        vsync: vga_vsync,
    );

    const MsgLen = 69;
    let timer = Timing::Timer<100_000_000>(clk);
    let timer2 = Timing::Timer<5_000_000>(clk);
    let messages = [
        "YODL; Yet anOther (hardware) Description Language by Nathan Soufflet!",
        "Visit https://github.com/nathsou/yodl for more information :)        ",
    ];

    let msg_index = Timing::Counter<1>(clk, en: timer.q, rst: 1'b0).q;
    let rand_color = LFSR::LFSR<32>(clk, enable: timer.q).q[8:0];
    let y = Timing::Counter<6>(clk, en: timer2.q);
    y.rst = y.q >= (Text::Cols - 1);

    let text = Text::TextDynamicPosition<MsgLen>(
        text: messages[msg_index],
        x_offset: 7'd5,
        y_offset: y.q,
        screen_x: vga.col,
        screen_y: vga.row,
    );

    let color = match vga.is_active_area and text.is_active {
        1'b0 => { r: 3'd0, g: 3'd0, b: 3'd0 },
        1'b1 => { r: rand_color[2:0], g: rand_color[5:3], b: rand_color[8:6] },
    };

    vga_red = color.r;
    vga_green = color.g;
    vga_blue = color.b;
    led = 4'd0;
}
